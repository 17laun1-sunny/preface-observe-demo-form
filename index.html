        async function apiGet(action, params = {}) {

            const query = new URLSearchParams({ action, ...params }).toString();

            const res = await fetch(`${APPS_SCRIPT_URL}?${query}`);

            return await res.json();

        }


        async function apiPost(data) {

            const res = await fetch(APPS_SCRIPT_URL, {

                method: "POST",

                headers: { "Content-Type": "text/plain;charset=utf-8" },

                body: JSON.stringify(data)

            });

            return await res.json();

        }

        function buildSummary() {

            const box = document.getElementById('summaryBox');

            let platforms = Array.from(document.querySelectorAll('.plat-chk:checked')).map(c => c.value);

            let html = `<p class="mb-1"><strong class="text-brandPrimary dark:text-white">Name:</strong> ${document.getElementById('name').value}</p>`;

            html += `<p class="mb-4"><strong class="text-brandPrimary dark:text-white">Email:</strong> ${document.getElementById('email').value}</p>`;

            

            platforms.forEach(p => {

                let sel = document.querySelector(`input[name="obs_${p}"]:checked`).value;

                if(sel === "skip") {

                    html += `<p class="mb-2"><strong class="text-brandPrimary dark:text-white">${p} Obs:</strong> <span class="text-gray-400 italic">Skipped</span></p>`;

                } else {

                    let oSlot = sysData.observations.find(o => o.id == sel);

                    html += `<p class="mb-2"><strong class="text-brandPrimary dark:text-white">${p} Obs:</strong> ${oSlot.display} w/ ${oSlot.teacher}</p>`;

                }

            });


            let demoSel = document.querySelector('input[name="demoSlot"]:checked').value;

            if(demoSel === "skip") {

                html += `<p class="mt-4"><strong class="text-brandPrimary dark:text-white">Teaching Demo:</strong> <span class="text-gray-400 italic">Skipped (Availability provided)</span></p>`;

            } else {

                let dSlot = sysData.demos.find(d => d.id === demoSel);

                html += `<p class="mt-4"><strong class="text-brandPrimary dark:text-white">Teaching Demo:</strong> ${dSlot.display}</p>`;

            }

            box.innerHTML = html;

        }


        function submitData() {

            document.getElementById('submitOverlay').classList.remove('hidden');


            let platforms = Array.from(document.querySelectorAll('.plat-chk:checked')).map(c => c.value);

            let obs = [];

            platforms.forEach(p => {

                let sel = document.querySelector(`input[name="obs_${p}"]:checked`).value;

                if (sel !== "skip") obs.push(sel);

            });

            let demoSel = document.querySelector('input[name="demoSlot"]:checked').value;


            const payload = {

                uuid: selectedUuid,

                name: document.getElementById('name').value,

                email: document.getElementById('email').value,

                platforms: platforms,

                observations: obs,

                demo: demoSel,

                demoAvailability: document.getElementById('demoAvailability').value

            };


            apiPost(payload).then(res => {

                document.getElementById('submitOverlay').classList.add('hidden');

                if (res.success) {

                    document.getElementById('bookingForm').classList.add('hidden');

                    document.getElementById('successView').classList.remove('hidden');

                    window.scrollTo({ top: 0, behavior: 'smooth' });

                } else {

                    alert(res.message);

                }

            }).catch(err => {

                document.getElementById('submitOverlay').classList.add('hidden');

                alert("An error occurred during submission. Please check your connection and try again.");

            });

        }
